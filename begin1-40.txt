print("Периметр:", perimeter_triangle)
print("Площадь:", area_triangle)

coefficient_a_quadratic = float(input("Введите число A: "))
coefficient_b_quadratic = float(input("Введите число B: "))
coefficient_c_quadratic = float(input("Введите число C: "))
discriminant = coefficient_b_quadratic ** 2 - 4 * coefficient_a_quadratic * coefficient_c_quadratic

# Проверка на наличие действительных корней
if discriminant < 0:
    print("Корней нет.")
else:
    sqrt_d = math.sqrt(discriminant)
    root_x1 = (-coefficient_b_quadratic - sqrt_d) / (2 * coefficient_a_quadratic)
    root_x2 = (-coefficient_b_quadratic + sqrt_d) / (2 * coefficient_a_quadratic)
    print("Меньший корень:", min(root_x1, root_x2))
    print("Больший корень:", max(root_x1, root_x2))
